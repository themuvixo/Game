cmake_minimum_required(VERSION 3.20)
project(AdventurePlatformer VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Official SFML 2.6.1 VC17 x64 prebuilt; pass -DSFML_DIR=<sfml>/lib/cmake/SFML
set(SFML_STATIC_LIBRARIES FALSE)
find_package(SFML 2.6.1 EXACT COMPONENTS system window graphics audio main REQUIRED)
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.cpp)
add_executable(AdventurePlatformer WIN32 ${GAME_SOURCES})
target_include_directories(AdventurePlatformer PRIVATE ${CMAKE_SOURCE_DIR}/src)
target_link_libraries(AdventurePlatformer PRIVATE sfml-graphics sfml-window sfml-system sfml-audio sfml-main)
if(MSVC)
  target_compile_options(AdventurePlatformer PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_compile_definitions(AdventurePlatformer PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
  set_property(TARGET AdventurePlatformer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
endif()
add_custom_command(TARGET AdventurePlatformer POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:AdventurePlatformer>/assets)
